// 1.2.3 updater checks (no network): version compare, release JSON, asset/checksum picking, URL rules, zip verification.
// python3 tools/update_fixtures.py /tmp/uf [a real release zip]
// mcs -nowarn:649 -out:/tmp/uh.exe -r:System.Web.Extensions.dll src/UpdateLogic.cs tools/UpdateHarness.cs.txt && mono /tmp/uh.exe /tmp/uf [latest.json]
using System;
using System.IO;
using System.Linq;
namespace BotBeacons
{
    static class UpdateHarness
    {
        static int cases, fails;
        static void Check(bool ok, string what) { cases++; if (!ok) { fails++; Console.WriteLine("  FAIL " + what); } }
        static void Main(string[] args)
        {
            string dir = args.Length > 0 ? args[0] : "/tmp/uf";
            // versions
            Check(UpdateLogic.Compare("v1.2.3", "1.2.3") == 0, "v prefix");
            Check(UpdateLogic.Compare("v1.2.4", "1.2.3") > 0 && UpdateLogic.Compare("1.2.3", "1.2.4") < 0, "patch");
            Check(UpdateLogic.Compare("1.10.0", "1.9.9") > 0, "numeric, not text (1.10 > 1.9)");
            Check(UpdateLogic.Compare("2.0", "1.99.99") > 0 && UpdateLogic.Compare("1.2", "1.2.0") == 0, "short versions");
            Check(UpdateLogic.Compare("1.2.3-beta", "1.2.3") < 0 && UpdateLogic.Compare("1.2.3", "1.2.3-rc.1") > 0, "pre-release is older than the release");
            Check(UpdateLogic.Compare("1.2.3-rc.2", "1.2.3-rc.10") < 0 && UpdateLogic.Compare("1.2.3-alpha", "1.2.3-beta") < 0, "pre-release order");
            Check(UpdateLogic.Compare("v1.1.3", "1.2.3") < 0, "the published v1.1.3 is older than 1.2.3 (this copy is newer than the latest)");
            Check(UpdateLogic.Compare("1.2.3+build.5", "1.2.3") == 0, "build metadata ignored");
            Check(UpdateLogic.Compare("nightly", "1.2.3") < 0 && UpdateLogic.Compare("", "1.0") < 0, "a tag that isn't a version never counts as newer");
            int[] n; string pre; Check(!UpdateLogic.TryParse("1.x.3", out n, out pre) && !UpdateLogic.TryParse("-1.2", out n, out pre), "junk rejected");

            // release JSON
            string mk = "{\"tag_name\":\"v1.3.0\",\"name\":\"1.3.0\",\"html_url\":\"https://github.com/SnazzyGaz/bot-beacons/releases/tag/v1.3.0\",\"draft\":false,\"prerelease\":false,\"assets\":[" +
                "{\"name\":\"BotBeacons-1.3.0-tabs-test-win.zip\",\"size\":10,\"browser_download_url\":\"https://github.com/SnazzyGaz/bot-beacons/releases/download/v1.3.0/BotBeacons-1.3.0-tabs-test-win.zip\"}," +
                "{\"name\":\"BotBeacons-1.3.0-win.zip\",\"size\":1733812,\"browser_download_url\":\"https://github.com/SnazzyGaz/bot-beacons/releases/download/v1.3.0/BotBeacons-1.3.0-win.zip\"}," +
                "{\"name\":\"SHA256SUMS\",\"size\":200,\"browser_download_url\":\"https://github.com/SnazzyGaz/bot-beacons/releases/download/v1.3.0/SHA256SUMS\"}," +
                "{\"name\":\"BotBeacons-1.3.0-win.zip.sha256\",\"size\":90,\"browser_download_url\":\"https://github.com/SnazzyGaz/bot-beacons/releases/download/v1.3.0/BotBeacons-1.3.0-win.zip.sha256\"}]}";
            var r = UpdateLogic.ParseRelease(mk);
            Check(r != null && r.Tag == "v1.3.0" && r.ZipName == "BotBeacons-1.3.0-win.zip" && r.ZipSize == 1733812, "release parsed, plain -win.zip preferred over the test build");
            Check(r != null && r.SumName == "BotBeacons-1.3.0-win.zip.sha256", "the zip's own .sha256 preferred over SHA256SUMS");
            var other = UpdateLogic.ParseRelease(mk.Replace("SnazzyGaz/bot-beacons/releases/download/v1.3.0/BotBeacons-1.3.0-win.zip\"", "evil.example/BotBeacons-1.3.0-win.zip\""));
            Check(other != null && other.ZipName == "BotBeacons-1.3.0-tabs-test-win.zip", "an asset off this repo's release path is never picked");
            Check(UpdateLogic.ParseRelease("{\"message\":\"API rate limit exceeded\"}") == null && UpdateLogic.ParseRelease("<html>") == null && UpdateLogic.ParseRelease("") == null, "rate-limit / junk answers are not releases");
            var none = UpdateLogic.ParseRelease("{\"tag_name\":\"v2.0.0\",\"assets\":[]}"); Check(none != null && none.ZipUrl == "", "a release without a Windows zip");
            if (args.Length > 1 && File.Exists(args[1]))
            {
                var real = UpdateLogic.ParseRelease(File.ReadAllText(args[1]));
                Check(real != null && real.ZipName.EndsWith("-win.zip") && real.ZipSize > 0 && UpdateLogic.AssetUrlOk(real.ZipUrl), "GitHub's real latest-release answer parses (" + (real != null ? real.Tag + " " + real.ZipName : "null") + ")");
                Check(real != null && UpdateLogic.Compare(real.Tag, "1.2.3") < 0, "1.2.3 is newer than the published latest (" + (real != null ? real.Tag : "") + "): reports up to date / newer");
            }

            // URLs
            Check(UpdateLogic.AssetUrlOk("https://github.com/SnazzyGaz/bot-beacons/releases/download/v1.3.0/x-win.zip"), "release asset URL ok");
            Check(!UpdateLogic.AssetUrlOk("http://github.com/SnazzyGaz/bot-beacons/releases/download/v1.3.0/x-win.zip") && !UpdateLogic.AssetUrlOk("https://github.com.evil.io/SnazzyGaz/bot-beacons/releases/download/v1/x.zip")
                && !UpdateLogic.AssetUrlOk("https://github.com/Someone/bot-beacons/releases/download/v1/x.zip") && !UpdateLogic.AssetUrlOk("https://github.com/SnazzyGaz/bot-beacons/archive/main.zip"), "everything else refused");
            Check(UpdateLogic.RedirectOk("https://objects.githubusercontent.com/x") && UpdateLogic.RedirectOk("https://release-assets.githubusercontent.com/x") && !UpdateLogic.RedirectOk("https://evil.example/x") && !UpdateLogic.RedirectOk("http://objects.githubusercontent.com/x") && !UpdateLogic.RedirectOk("https://githubusercontent.com.evil.io/x"), "redirects only to GitHub hosts over https");

            // checksums
            string h = new string('a', 64), h2 = new string('b', 64);
            Check(UpdateLogic.ParseChecksum(h + "  BotBeacons-1.3.0-win.zip\n" + h2 + "  other.zip\n", "BotBeacons-1.3.0-win.zip") == h, "sha256sum format");
            Check(UpdateLogic.ParseChecksum(h2 + " *other.zip\n" + h.ToUpperInvariant() + " *BotBeacons-1.3.0-win.zip\r\n", "BotBeacons-1.3.0-win.zip") == h, "binary-mode marker, CRLF, upper case");
            Check(UpdateLogic.ParseChecksum(h + "\n", "BotBeacons-1.3.0-win.zip") == h, "a lone hash (<zip>.sha256)");
            Check(UpdateLogic.ParseChecksum(h + "  a.zip\n" + h2 + "  b.zip\n", "BotBeacons-1.3.0-win.zip") == null, "a list that doesn't name the zip");
            Check(UpdateLogic.ParseChecksum("nope", "x") == null && UpdateLogic.ParseChecksum(null, "x") == null, "junk checksum file");

            // zips
            Func<string, byte[]> Z = f => File.ReadAllBytes(Path.Combine(dir, f));
            byte[] exe, good = Z("good.zip"), want = Z("exe.bin");
            Check(UpdateLogic.Verify(good, good.Length, UpdateLogic.Sha256(good), out exe) == null && exe != null && exe.SequenceEqual(want), "good zip: verified, the top-level BotBeacons.exe (not source/'s) extracted byte for byte");
            Check(UpdateLogic.Verify(good, -1, null, out exe) == null, "no size/checksum published: zip checks alone");
            Check(UpdateLogic.Verify(Z("stored.zip"), -1, null, out exe) == null && exe.SequenceEqual(want), "stored (uncompressed) entry");
            Check(Fails(good, good.Length + 1, null, "wrong size"), "wrong size refused");
            Check(Fails(good, good.Length, new string('0', 64), "SHA-256"), "checksum mismatch refused");
            foreach (var bad in new[] { "noexe.zip:no BotBeacons.exe", "onlysource.zip:no BotBeacons.exe", "notpe.zip:isn't a Windows program", "small.zip:isn't a Windows program", "crc.zip:damaged", "trunc.zip:not a valid zip", "notzip.zip:not a valid zip", "evil.zip:not a valid zip" })
            {
                var parts = bad.Split(':'); Check(Fails(Z(parts[0]), -1, null, parts[1]), parts[0] + " refused (" + parts[1] + ")");
            }
            Check(Fails(new byte[0], -1, null, "empty"), "empty download refused");
            if (File.Exists(Path.Combine(dir, "release.zip")))
            {
                var rz = Z("release.zip"); string e2 = UpdateLogic.Verify(rz, rz.Length, UpdateLogic.Sha256(rz), out exe);
                Check(e2 == null && exe != null && exe.Length > 100000, "a real Bot Beacons release zip verifies (" + (e2 ?? (exe.Length + " byte exe")) + ")");
            }
            // the helper batch
            Console.WriteLine(fails == 0 ? "OK: " + cases + " checks, 0 failures." : "FAILED: " + cases + " checks, " + fails + " failures.");
            Environment.Exit(fails == 0 ? 0 : 1);
        }
        static bool Fails(byte[] z, long size, string sha, string why) { byte[] e; var err = UpdateLogic.Verify(z, size, sha, out e); return err != null && err.Contains(why) && e == null; }
    }
}
